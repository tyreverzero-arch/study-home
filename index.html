const nums = [1, 2, 3, 4, 5];

2. 二分查找模板

function lowerBound(arr, target) {   // 第一个 >= target 的下标
  let l = 0, r = arr.length;
  while (l < r) {
    const mid = (l + r) >> 1;
    if (arr[mid] >= target) r = mid; else l = mid + 1;
  }
  return l;
}

3. 滑动窗口模板

def max_sub_no_repeat(s: str) -> int:
    win, l, ans = set(), 0, 0
    for r, ch in enumerate(s):
        while ch in win:          # 收缩左边界
            win.remove(s[l]); l += 1
        win.add(ch)
        ans = max(ans, r - l + 1)
    return ans

4. DP 套路

  • 先想状态定义，再想转移方程，最后定初始化与遍历方向
  • 背包：0-1 背包容量倒序，完全背包正序
  • 区间 DP：按区间长度枚举，dp[i][j] = min(dp[i][k] + dp[k+1][j] + cost)
